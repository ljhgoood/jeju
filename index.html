Power BI 권장 DAX

Avg Value =
AVERAGE(AirQuality[value])

Latest Year =
MAXX(
    ALLSELECTED(AirQuality[year]),
    AirQuality[year]
)

YoY % =
DIVIDE(
    [Current Avg] - [Previous Avg],
    [Previous Avg]
)

Station Rank =
RANKX(
    ALLSELECTED(AirQuality[station]),
    [Avg Value],
    ,
    DESC,
    Dense
)

Tableau 계산식

// Latest Year
{ FIXED : MAX([year]) }

// Current Avg
AVG(
    IF [year] = [Latest Year]
    THEN [value]
    END
)

// YoY %
([Current Avg] - [Previous Avg]) / [Previous Avg]

IF [YoY %] < 0 THEN "개선"
